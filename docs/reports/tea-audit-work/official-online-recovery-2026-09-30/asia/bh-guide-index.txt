function getWeatherData(lat, lon, lang = 'ar') {
          const apiKey = '231ab8924a185762323f4246df16544c'; // OpenWeather API key
          const apiUrl = `https://api.openweathermap.org/data/2.5/weather?lat=${lat}&lon=${lon}&appid=${apiKey}&units=metric&lang=${lang}`;
       
          // Log the API URL to the browser console (REMINDER: Remove this line later)
          console.log('API URL:', apiUrl);
       
          // Fetch weather data from OpenWeather API
          fetch(apiUrl)
            .then(response => response.json())
            .then(data => {
              const weatherDetailsContainer = document.getElementById('weather-widget-details');
              const locationContainer = document.getElementById('weather-widget-location');
       
              // Prepare weather details to display
              const location = `${data.name}, ${data.sys.country}`; // Country may still be in English
              const temperature = `${data.main.temp}°C`;
              const condition = data.weather[0].description;
              const iconCode = data.weather[0].icon; // OpenWeather icon code
       
              const weatherIcon = `<img src="http://openweathermap.org/img/wn/${iconCode}.png" alt="${condition}" class="weather-widget-icon">`;
       
              // Add data to the widget container
              locationContainer.innerHTML = location; // Display Location
              weatherDetailsContainer.innerHTML = `
                ${weatherIcon}
      <div class="weather-widget-temp">${temperature}</div>
      <div class="weather-widget-condition">${condition}</div>
              `;
            })
            .catch(err => {
              console.error('Failed to fetch weather data:', err);
              const weatherDetailsContainer = document.getElementById('weather-widget-details');
              const locationContainer = document.getElementById('weather-widget-location');
       
              locationContainer.innerHTML = 'Location not found';
              weatherDetailsContainer.innerHTML = '<p>Weather data is not available at this time.</p>';
            });
        }
       
        // Check for user's geolocation
        document.addEventListener('DOMContentLoaded', function() {
          if (navigator.geolocation) {
            navigator.geolocation.getCurrentPosition(function(position) {
              const lat = position.coords.latitude;
              const lon = position.coords.longitude;
       
              // Get weather data for the user's location
              getWeatherData(lat, lon, 'ar'); // Use 'ar' for Arabic language
            });
          } else {
            console.log('Geolocation is not supported by this browser.');
          }
        });

    function typeMessage(text, callback, sender = 'bot') {
      let index = 0;
      const msg = document.createElement('div');
      msg.className = `chat-message chat-${sender}`;
      chatbotMessages.appendChild(msg);

      function typeChar() {
        if (index < text.length) {
