function adjustFontSizes() {
            const isSmallScreen = $(window).width() <= 395;

            // Adjust font size for company name
            $('.company-name').each(function () {
                if (isSmallScreen) {
                    $(this).css('font-size', '0.8rem');
                } else {
                    $(this).css('font-size', '1.2rem');
                }
            });

            // Adjust font size for tagline
            $('.tagline').each(function () {
                if (isSmallScreen) {
                    $(this).css('font-size', '0.65rem');
                } else {
                    $(this).css('font-size', '13px');
                }
            });
        }

        // Run the function on page load and window resize
        $(document).ready(adjustFontSizes);
        $(window).resize(adjustFontSizes);
